from django.shortcuts import render, redirect, get_object_or_404
from django.contrib.auth.decorators import login_required
from django.http import Http404
from django.core.exceptions import PermissionDenied

from .models import Topic, Entry
from .forms import TopicForm, EntryForm


# ==================== 辅助函数 ====================

def check_topic_owner(topic, request):
    """确认主题属于当前用户"""
    if topic.owner != request.user:
        raise PermissionDenied("这不是你的主题，你没有权限进行此操作。")


def get_topic_or_404(username, category_path):
    """根据 username 和 category_path 匹配唯一的 Topic，并校验路径完整性"""
    slugs = [s for s in category_path.strip('/').split('/') if s]
    if not slugs:
        raise Http404("分类路径不能为空")

    target_slug = slugs[-1]

    # 1. 获取对应用户下目标 slug 的 Topic
    topic = get_object_or_404(Topic, owner__username=username, slug=target_slug)

    # 2. 校验路径是否完全匹配（防伪造 URL，如把 /tech/python 错拼成 /other/python）
    if topic.get_full_path() != category_path.strip('/'):
        raise Http404("分类路径不匹配")

    return topic


# ==================== 视图函数 ====================

def index(request):
    """学习笔记的主页"""
    return render(request, 'learning_logs/index.html')


@login_required
def topics(request, username=None):
    """显示指定用户（默认当前用户）的所有主题"""
    target_username = username or request.user.username
    
    # 结合 MPTT：可以获取该用户下的所有分类，在前端使用 {% recursetree %} 渲染成树状图
    topics = Topic.objects.filter(owner__username=target_username)
    context = {'topics': topics, 'target_username': target_username}
    return render(request, 'learning_logs/topics.html', context)


@login_required
def topic(request, username, category_path):
    """显示单个主题及其所有的条目（包含子主题条目）"""
    topic = get_topic_or_404(username, category_path)

    # 利用 MPTT 的 get_descendants 一键查出当前分类及所有子孙分类的条目
    sub_topics = topic.get_descendants(include_self=True)
    entries = Entry.objects.filter(topic__in=sub_topics).order_by('-date_added')

    context = {
        'topic': topic, 
        'entries': entries,
        'username': username
    }
    return render(request, 'learning_logs/topic.html', context)


@login_required
def new_topic(request):
    """添加新主题"""
    if request.method != 'POST':
        # 实例化表单时传入 user，方便表单过滤父分类列表
        form = TopicForm(user=request.user)
    else:
        form = TopicForm(data=request.POST, user=request.user)
        if form.is_valid():
            new_topic = form.save(commit=False)
            new_topic.owner = request.user
            new_topic.save()
            return redirect(
                'learning_logs:topic', 
                username=request.user.username, 
                category_path=new_topic.get_full_path()
            )

    context = {'form': form}
    return render(request, 'learning_logs/new_topic.html', context)


@login_required
def edit_topic(request, username, category_path):
    """编辑既有的主题名称或层级关系"""
    topic = get_topic_or_404(username, category_path)
    check_topic_owner(topic, request)

    if request.method != 'POST':
        form = TopicForm(instance=topic, user=request.user)
    else:
        form = TopicForm(instance=topic, data=request.POST, user=request.user)
        if form.is_valid():
            updated_topic = form.save()
            # 修改父节点或名称后，URL 路径可能发生改变，重定向到最新生成的完整路径
            return redirect(
                'learning_logs:topic', 
                username=username, 
                category_path=updated_topic.get_full_path()
            )

    context = {'topic': topic, 'form': form}
    return render(request, 'learning_logs/edit_topic.html', context)


@login_required
def delete_topic(request, username, category_path):
    """删除既有的主题（包括其子主题）"""
    topic = get_topic_or_404(username, category_path)
    check_topic_owner(topic, request)

    if request.method == 'POST':
        topic.delete()
        return redirect('learning_logs:topics', username=username)

    context = {'topic': topic}
    return render(request, 'learning_logs/delete_topic.html', context)


@login_required
def new_entry(request, username, category_path):
    """在特定主题中添加新条目"""
    topic = get_topic_or_404(username, category_path)
    check_topic_owner(topic, request)

    if request.method != 'POST':
        form = EntryForm()
    else:
        form = EntryForm(data=request.POST)
        if form.is_valid():
            new_entry = form.save(commit=False)
            new_entry.topic = topic
            new_entry.save()
            return redirect(
                'learning_logs:topic', 
                username=username, 
                category_path=topic.get_full_path()
            )

    context = {'topic': topic, 'form': form}
    return render(request, 'learning_logs/new_entry.html', context)


@login_required
def edit_entry(request, entry_id):
    """编辑既有的条目"""
    entry = get_object_or_404(Entry, id=entry_id)
    topic = entry.topic
    check_topic_owner(topic, request)

    if request.method != 'POST':
        form = EntryForm(instance=entry)
    else:
        form = EntryForm(instance=entry, data=request.POST)
        if form.is_valid():
            form.save()
            return redirect(
                'learning_logs:topic', 
                username=topic.owner.username, 
                category_path=topic.get_full_path()
            )

    context = {'entry': entry, 'topic': topic, 'form': form}
    return render(request, 'learning_logs/edit_entry.html', context)


@login_required
def delete_entry(request, entry_id):
    """删除既有的条目"""
    entry = get_object_or_404(Entry, id=entry_id)
    topic = entry.topic
    check_topic_owner(topic, request)

    if request.method == 'POST':
        entry.delete()

    return redirect(
        'learning_logs:topic', 
        username=topic.owner.username, 
        category_path=topic.get_full_path()
    )
